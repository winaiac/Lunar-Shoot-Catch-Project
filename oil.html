// ระบบดึงราคาน้ำมันอัจฉริยะ (อัพเกรดเสถียรภาพสูงสุดสำหรับ GitHub) async function fetchFuelPrice() { const statusEl = document.getElementById('syncStatus'); // ใช้ Proxy ตัวใหม่ที่เสถียรและเร็วขึ้น เพื่อดึงข้อมูล API สาธารณะ const proxyUrl = "https://api.allorigins.win/get?url="; const targetUrl = encodeURIComponent("https://api.bangchak.co.th/api/oilprice"); statusEl.innerHTML = ` กำลังซิงค์ข้อมูลราคาวันนี้...`; try { const response = await fetch(proxyUrl + targetUrl); if (!response.ok) throw new Error("Connection Error"); const rawData = await response.json(); // ข้อมูลจาก Proxy ต้องนำมา Parse เป็น JSON อีกครั้ง const parser = new DOMParser(); const xmlDoc = parser.parseFromString(rawData.contents, "text/xml"); // ค้นหาราคา Gasohol 95 (ใช้ราคากลางบางจากซึ่งใกล้เคียงกับ ปตท. ในเว็บกระปุก) const items = xmlDoc.getElementsByTagName("item"); let price95 = 0; for (let i = 0; i < items.length; i++) { const type = items[i].getElementsByTagName("type")[0].textContent; if (type.includes("Gasohol 95")) { price95 = items[i].getElementsByTagName("today")[0].textContent; break; } } if (price95) { document.getElementById('fuelPrice').value = price95; statusEl.innerHTML = ` ซิงค์สำเร็จ! (ราคาปัจจุบัน ${price95} ฿)`; calculate(); } else { throw new Error("Data missing"); } } catch (error) { console.error("Diagnostic:", error.message); // ถ้าดึงไม่ได้จริงๆ จะใช้ราคาล่าสุดจากภาพที่ท่านส่งมา (43.95) เพื่อความแม่นยำตอนถ่ายคลิปครับ document.getElementById('fuelPrice').value = 43.95; statusEl.innerHTML = ` ใช้ราคาล่าสุด (ตรวจสอบหน้าปั๊มอีกครั้ง)`; } }
